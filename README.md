<script>
  // small helpers
  document.getElementById('year').textContent = new Date().getFullYear();

  // Accordion
  function toggleAccordion(btn){
    const expanded = btn.getAttribute('aria-expanded') === 'true';
    btn.setAttribute('aria-expanded', String(!expanded));
    const answer = btn.parentElement.querySelector('.answer');
    if(!expanded){
      answer.hidden = false;
      btn.querySelector('.small').textContent = '−';
    } else {
      answer.hidden = true;
      btn.querySelector('.small').textContent = '+';
    }
  }

  // Modal controls
  const modal = document.getElementById('modal');
  function openModal(){
    modal.classList.add('show');
    modal.setAttribute('aria-hidden','false');
  }
  function closeModal(){
    modal.classList.remove('show');
    modal.setAttribute('aria-hidden','true');
  }

  // Quick booking: open WhatsApp with prefilled message
  function handleQuickBook(e){
    e.preventDefault();
    const name = encodeURIComponent(document.getElementById('customerName').value.trim());
    const phone = encodeURIComponent(document.getElementById('customerPhone').value.trim());
    const pickup = encodeURIComponent(document.getElementById('pickup').value.trim());
    const carType = encodeURIComponent(document.getElementById('carType').value);
    if(!name || !phone || !pickup || !carType){
      alert('Please fill all required fields.');
      return;
    }
    // Create prefilled WhatsApp message
    let msg = `Hello AAZ Car Rental,%0A%0AMy name is ${name}.%0AMobile: ${phone}.%0APick-up: ${pickup}.%0ARequested car: ${carType}.%0A%0APlease share availability and price.`;
    const waNumber = '971501466299'; // +971501466299 (no leading 0)
    const url = `https://wa.me/${waNumber}?text=${msg}`;
    window.open(url, '_blank');
  }

  // Full booking form -> prefilled WhatsApp message
  function handleFullBooking(e){
    e.preventDefault();
    const name = encodeURIComponent(document.getElementById('fullName').value.trim());
    const phone = encodeURIComponent(document.getElementById('fullPhone').value.trim());
    const pickup = encodeURIComponent(document.getElementById('fullPickup').value.trim());

    const date = encodeURIComponent(document.getElementById('pickupDate').value);
    const message = encodeURIComponent(document.getElementById('customerMessage').value.trim());

    if(!name || !phone || !pickup || !carType || !date){
      alert('Please fill all required fields.');
      return;
    }
    let msg = `Hello AAZ Car Rental,%0A%0AMy name is ${name}.%0AMobile: ${phone}.%0APick-up: ${pickup}.%0ADate: ${date}.%0ARequested car: ${carType}.%0AMessage: ${message}%0A%0APlease share rates and next steps.`;
    const waNumber = '971501466299';
    const url = `https://wa.me/${waNumber}?text=${msg}`;
    window.open(url, '_blank');
  }

  // Smooth scroll for internal links (header nav)
  document.querySelectorAll('a[href^="#"]').forEach(a=>{
    a.addEventListener('click', function(e){
      const target = document.querySelector(this.getAttribute('href'));
      if(target){
        e.preventDefault();
        target.scrollIntoView({behavior:'smooth',block:'start'});
      }
    });
  });
